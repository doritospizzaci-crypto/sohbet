Firebase nasıl kurulur?
  1. console.firebase.google.com → proje oluştur
  2. Build → Realtime Database → Create database
  3. Rules sekmesine aşağıdaki kuralları yapıştırıp Publish et
  4. Data sekmesindeki https://… adresini yukarıya yaz
{
  "rules": {
    "r": {
      "$room": {
        ".read": true,
        ".write": true,
        "$msg": {
          ".validate": "newData.hasChildren(['iv','c','ts'])",
          "iv": {".validate": "newData.isString() && newData.val().length < 40"},
          "c": {".validate": "newData.isString() && newData.val().length < 3000000"},
          "ts": {".validate": "newData.isNumber()"},
          "$other": {".validate": false}
        }
      }
    }
  }
}
